              products: [...prev.products, ...chunk]
            }));
          });

      this.observer.disconnect();

    const lqipUrl = this.props.src.replace('.jpg', '_thumbnail.jpg');
    
    img.src = lqipUrl;

      this.setState({ thumbnailLoaded: true });
      
      // 再加载高质量图
      const hqImg = new Image();
      hqImg.src = this.props.src;
      hqImg.onload = () => {
        this.setState({ loaded: true });
      };
      hqImg.onerror = () => {
        this.setState({ error: true });
      };
    };
  };
  
  render() {
    return (
      <div className="image-container" ref={this.imgRef}>
        {/* 骨架屏 */}
        {!this.state.thumbnailLoaded && <div className="image-skeleton" />}
        
        {/* 低质量占位图 */}
        {this.state.thumbnailLoaded && !this.state.loaded && (
          <img

            className="blur-up"

        {this.state.error && (

        )}
      </div>
    );
  }
}

class ProductFilterSystem {
  constructor() {
    // 筛选条件状态
    this.state = {
      filters: {
        category: [],      // 多选分类
        priceRange: { min: 0, max: 10000 },  // 价格区间
        brand: [],         // 品牌多选
        rating: 0,         // 最低评分
        attributes: {}     // 动态属性筛选
      },
      sortBy: 'recommended',  // 排序方式
      keyword: '',           // 搜索关键词
      page: 1,
      pageSize: 50
    };
    
    // 防抖和缓存
    this.searchDebounceTimer = null;
    this.filterCache = new Map();
    this.urlState = new URLStateManager();
  }
  
  // 防抖搜索（300ms）
  handleSearch = debounce((keyword) => {
    this.setState({ keyword, page: 1 });
    this.syncToURL();
    this.fetchProducts();
  }, 300);
  
  // 筛选条件变更
  handleFilterChange = (filterType, value) => {
    // 批量更新筛选条件
    this.setState(prev => ({
      filters: {
        ...prev.filters,
        [filterType]: value
      },
      page: 1  // 重置页码
    }));
    
    // 使用requestAnimationFrame避免频繁重渲染
    requestAnimationFrame(() => {
      this.fetchProducts();
    });
  };
  
  // 构建查询参数
  buildQueryParams() {
    const params = new URLSearchParams();
    
    // 筛选条件
    Object.entries(this.state.filters).forEach(([key, value]) => {
      if (Array.isArray(value) && value.length > 0) {
        params.append(key, value.join(','));
      } else if (typeof value === 'object' && value !== null) {
        params.append(`${key}_min`, value.min);
        params.append(`${key}_max`, value.max);
      } else if (value) {
        params.append(key, value);
      }
    });
    
    // 排序
    params.append('sort', this.state.sortBy);
    
    // 分页
    params.append('page', this.state.page);
    params.append('pageSize', this.state.pageSize);
    
    // 搜索
    if (this.state.keyword) {
      params.append('q', this.state.keyword);
    }
    
    return params;
  }
  
  // 请求商品数据
  async fetchProducts() {
    const cacheKey = this.buildQueryParams().toString();
    
    // 检查缓存
    if (this.filterCache.has(cacheKey)) {
      const cached = this.filterCache.get(cacheKey);
      if (Date.now() - cached.timestamp < 300000) { // 5分钟缓存
        return cached.data;
      }
    }
    
    // 显示加载状态
    this.setState({ loading: true });
    
    try {
      // 使用AbortController取消旧请求
      if (this.controller) {
        this.controller.abort();
      }
      this.controller = new AbortController();
      
      const response = await fetch(`/api/products?${cacheKey}`, {
        signal: this.controller.signal,
        priority: 'high'  // 高优先级
      });
      
      const data = await response.json();
      
      // 缓存结果
      this.filterCache.set(cacheKey, {
        data,
        timestamp: Date.now()
      });
      
      // 清理旧缓存（最多保存100条）
      if (this.filterCache.size > 100) {
        const oldestKey = Array.from(this.filterCache.keys())[0];
        this.filterCache.delete(oldestKey);
      }
      
      return data;
    } catch (error) {
      if (error.name !== 'AbortError') {
        console.error('Failed to fetch products:', error);
        // 降级到本地缓存
        return this.getFallbackData();
      }
    } finally {
      this.setState({ loading: false });
    }
  }
  
  // URL状态同步（保持浏览器前进后退）
  syncToURL() {
    const params = this.buildQueryParams();
    window.history.replaceState(
      null,
      '',
      `${window.location.pathname}?${params}`
    );
  }
}

class ProductWorkerManager {
  constructor() {
    // 创建Worker处理筛选、排序等计算密集型任务
    this.filterWorker = new Worker('./productFilter.worker.js');
    this.sortWorker = new Worker('./productSort.worker.js');
    
    this.workerCallbacks = new Map();
    this.callbackId = 0;
  }
  
  // 在Worker中筛选商品
  filterProductsInWorker(products, filters) {
    return new Promise((resolve) => {
      const id = ++this.callbackId;
      
      this.workerCallbacks.set(id, resolve);
      
      this.filterWorker.postMessage({
        type: 'FILTER',
        id,
        payload: {
          products,
          filters
        }
      });
    });
  }
  
  // 在Worker中排序
  sortProductsInWorker(products, sortBy) {
    return new Promise((resolve) => {
      const id = ++this.callbackId;
      
      this.workerCallbacks.set(id, resolve);
      
      this.sortWorker.postMessage({
        type: 'SORT',
        id,
        payload: {
          products,
          sortBy
        }
      });
    });
  }
}

self.addEventListener('message', (e) => {
  const { type, id, payload } = e.data;
  
  if (type === 'FILTER') {
    const { products, filters } = payload;
    
    // 执行复杂的筛选逻辑
    const filtered = products.filter(product => {
      // 价格筛选
      if (product.price < filters.priceRange.min || 
          product.price > filters.priceRange.max) {
        return false;
      }
      
      // 分类筛选
      if (filters.category.length > 0 && 
          !filters.category.includes(product.category)) {
        return false;
      }
      
      // 品牌筛选
      if (filters.brand.length > 0 && 
          !filters.brand.includes(product.brand)) {
        return false;
      }
      
      // 评分筛选
      if (product.rating < filters.rating) {
        return false;
      }
      
      // 属性筛选
      if (filters.attributes) {
        for (const [key, value] of Object.entries(filters.attributes)) {
          if (product.attributes[key] !== value) {
            return false;
          }
        }
      }
      
      return true;
    });
    
    self.postMessage({ id, result: filtered });
  }
});

class ProductMemoryManager {
  constructor() {
    this.productCache = new Map();
    this.imageCache = new Map();
    this.maxCacheSize = 1000;
  }
  
  // LRU缓存策略
  addToCache(key, data) {
    if (this.productCache.size >= this.maxCacheSize) {
      // 删除最久未使用的
      const oldestKey = this.productCache.keys().next().value;
      this.productCache.delete(oldestKey);
    }
    
    this.productCache.set(key, {
      data,
      timestamp: Date.now(),
      accessCount: 0
    });
  }
  
  // 定期清理
  startCleanup() {
    setInterval(() => {
      const now = Date.now();
      const oneHour = 60 * 60 * 1000;
      
      for (const [key, item] of this.productCache) {
        if (now - item.timestamp > oneHour && item.accessCount === 0) {
          this.productCache.delete(key);
        }
      }
    }, 5 * 60 * 1000); // 每5分钟清理一次
  }
  
  // 图片预加载
  preloadImages(imageUrls) {
    imageUrls.forEach(url => {
      if (!this.imageCache.has(url)) {
        const img = new Image();
        img.src = url;
        this.imageCache.set(url, img);
      }
    });
  }
}

      return this.getPopularSearches();
    }
    
    // 检查缓存
    const cacheKey = `suggest:${keyword}`;
    if (this.suggestionCache.has(cacheKey)) {
      return this.suggestionCache.get(cacheKey);
    }
    
    // 多维度搜索建议
    const [products, categories, brands] = await Promise.all([
      this.searchProducts(keyword, { limit: 5 }),
      this.searchCategories(keyword),
      this.searchBrands(keyword)
    ]);
    
    const suggestions = {
      products: products.map(p => ({
        type: 'product',
        title: p.name,
        subtitle: `¥${p.price}`,
        image: p.image,
        url: `/product/${p.id}`
      })),
      categories: categories.map(c => ({
        type: 'category',
        title: c.name,
        count: c.productCount,
        url: `/category/${c.id}`
      })),
      brands: brands.map(b => ({
        type: 'brand',
        title: b.name,
        logo: b.logo,
        url: `/brand/${b.id}`
      })),
      keywords: this.generateKeywordSuggestions(keyword)
    };
    
    // 缓存结果
    this.suggestionCache.set(cacheKey, suggestions);
    
    return suggestions;
  }, 200);
  
  // 高级搜索功能
  advancedSearch = async (params) => {
    const {
      keyword,
      filters = {},
      sortBy = 'relevance',
      page = 1,
      pageSize = 50
    } = params;
    
    // 构建Elasticsearch查询
    const esQuery = {
      query: {
        bool: {
          must: [
            {
              multi_match: {
                query: keyword,
                fields: [
                  'name^3',    // 名称权重更高
                  'description^2',
                  'category^2',
                  'brand',
                  'attributes'
                ],
                fuzziness: 'AUTO'  // 模糊匹配
              }
            }
          ],
          filter: this.buildFilters(filters)
        }
      },
      sort: this.buildSort(sortBy),
      from: (page - 1) * pageSize,
      size: pageSize,
      highlight: {
        fields: {
          name: {},
          description: {}
        }
      },
      aggs: this.buildAggregations(filters) // 聚合统计
    };
    
    // 执行搜索
    const response = await fetch('/api/search', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(esQuery)
    });
    
    const result = await response.json();
    
    // 处理高亮显示
    const highlightedProducts = result.hits.hits.map(hit => ({
      ...hit._source,
      highlights: hit.highlight
    }));
    
    return {
      products: highlightedProducts,
      total: result.hits.total.value,
      aggregations: result.aggregations,
      took: result.took
    };
  };
  
  // 构建筛选条件
  buildFilters(filters) {
    const filterQueries = [];
    
    if (filters.priceRange) {
      filterQueries.push({
        range: {
          price: {
            gte: filters.priceRange.min,
            lte: filters.priceRange.max
          }
        }
      });
    }
    
    if (filters.category && filters.category.length > 0) {
      filterQueries.push({
        terms: {
          category_id: filters.category
        }
      });
    }
    
    if (filters.brand && filters.brand.length > 0) {
      filterQueries.push({
        terms: {
          brand_id: filters.brand
        }
      });
    }
    
    return filterQueries;
  }
  
  // 搜索联想词
  generateKeywordSuggestions(keyword) {
    const commonCombinations = [
      `${keyword} 优惠`,
      `${keyword} 新款`,
      `${keyword} 2024`,
      `${keyword} 折扣`
    ];
    
    // 基于历史搜索的联想
    const fromHistory = this.searchHistory
      .filter(hist => hist.includes(keyword) && hist !== keyword)
      .slice(0, 5);
    
    return [...commonCombinations, ...fromHistory];
  }
}

class CollaborativeEditorClient {
  constructor() {
    // 1. 编辑器实例
    this.editor = new ProseMirrorEditor();
    
    // 2. 本地状态管理
    this.localState = {
      revision: 0,                    // 当前版本号
      pendingOperations: [],          // 待确认操作队列
      buffer: [],                     // 离线缓冲操作
      cursorPosition: null,           // 本地光标位置
      selection: null                 // 本地选择范围
    };
    
    // 3. 远程用户状态
    this.remoteUsers = new Map();     // key: userId, value: {cursor, selection, color, name}
    
    // 4. 通信层
    this.socket = new RealtimeSocket();
    this.operationManager = new OperationManager();
    this.conflictResolver = new OTConflictResolver();
    
    // 5. 初始化
    this.setupEventListeners();
    this.connectToServer();
  }
  
  setupEventListeners() {
    // 监听本地编辑事件
    this.editor.on('change', (transaction) => {
      const operation = this.transformToOperation(transaction);
      this.handleLocalOperation(operation);
    });
    
    // 监听光标/选择变化
    this.editor.on('selectionChange', (selection) => {
      this.updateCursorPosition(selection);
      this.broadcastCursorPosition();
    });
    
    // 接收远程操作
    this.socket.on('operation', (operation) => {
      this.handleRemoteOperation(operation);
    });
    
    // 接收光标更新
    this.socket.on('cursorUpdate', (data) => {
      this.updateRemoteCursor(data);
    });
    
    // 处理断线重连
    this.socket.on('reconnect', () => {
      this.synchronizeState();
    });
  }
  
  // 本地操作处理流程
  handleLocalOperation(operation) {
    // 1. 乐观更新（立即应用）
    this.editor.applyOperation(operation);
    
    // 2. 添加到待确认队列
    this.localState.pendingOperations.push({
      operation,
      timestamp: Date.now(),
      revision: this.localState.revision
    });
    
    // 3. 发送到服务器
    this.socket.emit('operation', {
      operation,
      revision: this.localState.revision,
      clientId: this.getClientId()
    });
    
    // 4. 更新本地版本号
    this.localState.revision++;
  }
  
  // 远程操作处理
  async handleRemoteOperation(remoteOp) {
    // 1. 转换操作（解决冲突）
    const transformedOp = await this.conflictResolver.transform(
      remoteOp.operation,
      this.localState.pendingOperations
    );
    
    // 2. 应用到编辑器
    this.editor.applyOperation(transformedOp);
    
    // 3. 更新待确认队列中的操作
    this.updatePendingOperations(transformedOp);
    
    // 4. 确认收到操作
    this.socket.emit('ack', {
      operationId: remoteOp.id,
      revision: this.localState.revision
    });
  }
  
  // 光标位置同步
  updateCursorPosition(selection) {
    this.localState.cursorPosition = {
      from: selection.from,
      to: selection.to,
      timestamp: Date.now()
    };
  }
  
  broadcastCursorPosition() {
    // 节流发送光标更新
    this.throttle(() => {
      this.socket.emit('cursor', {
        position: this.localState.cursorPosition,
        userId: this.getUserId()
      });
    }, 100);
  }
  
  updateRemoteCursor(data) {
    const { userId, position } = data;
    
    if (!this.remoteUsers.has(userId)) {
      this.remoteUsers.set(userId, {
        cursor: position,
        color: this.generateUserColor(userId),
        name: `用户${userId.slice(0, 4)}`
      });
    } else {
      const user = this.remoteUsers.get(userId);
      user.cursor = position;
      user.lastSeen = Date.now();
    }
    
    // 渲染远程光标
    this.renderRemoteCursors();
  }
  
  // 渲染远程光标和选择范围
  renderRemoteCursors() {
    this.remoteUsers.forEach((user, userId) => {
      if (Date.now() - user.lastSeen > 30000) {
        // 30秒无活动，移除
        this.remoteUsers.delete(userId);
        return;
      }
      
      // 创建或更新光标标记
      this.editor.decorateRemoteCursor(userId, user.cursor, user.color);
    });
  }
}

        socket.close(4001, 'Access denied');
        return;
      }
      
      // 加入文档房间
      this.joinRoom(socket, docId, userId);
      
      // 发送当前文档状态
      const docState = await this.getDocumentState(docId);
      socket.emit('init', docState);
      
      // 通知其他用户新用户加入
      this.broadcastToRoom(docId, 'userJoined', { userId });
    });
    
    socket.on('operation', async (data) => {
      const { docId, operation, revision, clientId } = data;
      
      try {
        // 1. 验证操作有效性
        const isValid = this.validateOperation(operation, revision);
        if (!isValid) {
          socket.emit('error', { message: 'Invalid operation' });
          return;
        }
        
        // 2. 应用操作转换
        const transformedOp = await this.otEngine.transform(
          docId,
          operation,
          revision
        );
        
        // 3. 更新文档状态
        await this.applyOperation(docId, transformedOp);
        
        // 4. 广播给其他用户
        this.broadcastToRoom(docId, 'operation', {
          operation: transformedOp,
          clientId,
          revision: transformedOp.revision,
          timestamp: Date.now()
        }, socket); // 排除发送者
        
        // 5. 存储操作日志
        await this.storeOperationLog(docId, transformedOp);
        
        // 6. 确认操作
        socket.emit('ack', { 
          operationId: data.operationId,
          revision: transformedOp.revision 
        });
        
      } catch (error) {
        console.error('Operation processing error:', error);
        socket.emit('error', { message: 'Operation failed' });
      }
    });
    
    socket.on('cursor', (data) => {
      const { docId, position, userId } = data;
      // 广播光标更新（排除自己）
      this.broadcastToRoom(docId, 'cursorUpdate', {
        userId,
        position,
        timestamp: Date.now()
      }, socket);
    });
    
    socket.on('disconnect', () => {
      this.handleDisconnect(socket);
    });
  }
  
  // 操作转换引擎
  class OTEngine {
    constructor() {
      this.operationLogs = new Map(); // key: docId, value: array of operations
    }
    
    async transform(docId, incomingOp, clientRevision) {
      const logs = this.operationLogs.get(docId) || [];
      const serverRevision = logs.length;
      
      if (clientRevision < serverRevision) {
        // 客户端版本落后，需要转换
        const pendingOps = logs.slice(clientRevision);
        
        // 应用OT算法转换
        let transformedOp = incomingOp;
        for (const pendingOp of pendingOps) {
          transformedOp = this.otTransform(transformedOp, pendingOp);
        }
        
        return {
          ...transformedOp,
          revision: serverRevision + 1
        };
      } else if (clientRevision === serverRevision) {
        // 版本一致，直接应用
        return {
          ...incomingOp,
          revision: serverRevision + 1
        };
      } else {
        throw new Error('Client revision ahead of server');
      }
    }
    
    // OT转换函数（核心算法）
    otTransform(op1, op2) {
      // 根据操作类型进行转换
      // 这里简化实现，实际需要处理插入、删除、格式化等
      if (op1.type === 'insert' && op2.type === 'insert') {
        return this.transformInsertInsert(op1, op2);
      } else if (op1.type === 'insert' && op2.type === 'delete') {
        return this.transformInsertDelete(op1, op2);
      } else if (op1.type === 'delete' && op2.type === 'insert') {
        return this.transformDeleteInsert(op1, op2);
      } else if (op1.type === 'delete' && op2.type === 'delete') {
        return this.transformDeleteDelete(op1, op2);
      }
      
      return op1;
    }
    
    transformInsertInsert(op1, op2) {
      // 两个插入操作的转换
      // 如果op1位置在op2之前，不变
      // 如果op1位置在op2之后，位置需要调整
      if (op1.position <= op2.position) {
        return op1;
      } else {
        return {
          ...op1,
          position: op1.position + op2.text.length
        };
      }
    }
  }
}

// OT算法的核心实现
class OperationalTransformer {
  // 1. 操作定义
  static OperationTypes = {
    INSERT: 'insert',
    DELETE: 'delete',
    FORMAT: 'format',
    MOVE: 'move'
  };
  
  // 2. 位置转换（核心）
  static transformPosition(position, operation, isOwnOperation = false) {
    switch (operation.type) {
      case 'insert':
        if (position >= operation.position) {
          return position + operation.text.length;
        }
        return position;
        
      case 'delete':
        if (position > operation.position + operation.length) {
          return position - operation.length;
        } else if (position > operation.position) {
          // 位置在删除范围内
          return operation.position;
        }
        return position;
        
      default:
        return position;
    }
  }
  
  // 3. 操作转换（TP2算法）
  static transformOperation(op1, op2, isOwnOp1 = false) {
    // TP2算法确保交换律：transform(transform(op1, op2), op2') = transform(transform(op1, op2'), op2)
    
    if (op1.type === 'insert' && op2.type === 'insert') {
      if (op1.position < op2.position ||
          (op1.position === op2.position && isOwnOp1)) {
        return op1;
      } else {
        return {
          ...op1,
          position: op1.position + op2.text.length
        };
      }
    }
    
    if (op1.type === 'insert' && op2.type === 'delete') {
      if (op1.position <= op2.position) {
        return op1;
      } else if (op1.position > op2.position + op2.length) {
        return {
          ...op1,
          position: op1.position - op2.length
        };
      } else {
        // 插入位置在删除范围内
        return {
          ...op1,
          position: op2.position
        };
      }
    }
    
    if (op1.type === 'delete' && op2.type === 'insert') {
      if (op1.position < op2.position) {
        return op1;
      } else {
        return {
          ...op1,
          position: op1.position + op2.text.length
        };
      }
    }
    
    if (op1.type === 'delete' && op2.type === 'delete') {
      // 复杂情况：需要处理重叠删除
      return this.transformDeleteDelete(op1, op2);
    }
    
    return op1;
  }
}

      a.timestamp - b.timestamp

### **2. 性能优化难点**

- ✅ **安全性和权限**：实时权限验证、操作验证、审计日志

## 总结

**核心设计原则**：
1. **最终一致性**：使用OT确保所有客户端最终状态一致
2. **本地优先**：乐观更新提升用户体验
3. **渐进增强**：支持从简单到复杂的编辑功能
4. **可扩展性**：架构支持从几十到数百万用户

**关键技术决策**：
- ✅ **冲突解决**：推荐OT算法，成熟稳定
- ✅ **编辑器**：推荐ProseMirror，架构优秀
- ✅ **通信**：WebSocket + 操作确认机制
- ✅ **存储**：Redis（实时）+ PostgreSQL（持久化）

**最关键的技术难点**：
1. **操作冲突解决**：必须100%正确，否则数据会损坏
2. **光标位置同步**：需要精确的数学计算
3. **离线同步**：必须处理好所有边缘情况
4. **大规模并发**：架构必须有良好的扩展性

**最后建议**：从简单开始，逐步增加复杂性，充分测试每个功能，特别是冲突解决和离线同步。优先保证核心功能的正确性，再优化性能和用户体验。
